# 三条系统链 — 流程大纲（改完告诉我，我填细节）
# 格式: node_id [类型:格式] → 后继
# 类型: 🧠推理 🔀路由 📝写入 ❓提问
# 缩进表示层级/循环

═══════════════════════════════════════
🗺️ 地图链
═══════════════════════════════════════
m_init_plan         [🧠] → m_init_todos
m_init_todos        [📝→chain_todos] → m_world_check

  world 大千世界与宇宙:
    m_world_check   [🔀] → YES→m_world_plan | NO→m_planet_check
    m_world_plan    [🧠:location] → m_world_build
    m_world_build   [📝→location] → m_world_verify
    m_world_verify  [🧠] → CONTINUE→m_world_build | DONE→m_planet_check

  planet 星球与大陆:
    m_planet_check  [🔀] → YES→m_planet_plan | NO→m_nation_check
    m_planet_plan   [🧠:location] → m_planet_build
    m_planet_build  [📝→location] → m_planet_verify
    m_planet_verify [🧠] → CONTINUE→m_planet_build | DONE→m_nation_check

  nation 国家:
    m_nation_check  [🔀] → YES→m_nation_plan | NO→m_city_check
    m_nation_plan   [🧠:location] → m_nation_build
    m_nation_build  [📝→location] → m_nation_verify
    m_nation_verify [🧠] → CONTINUE→m_nation_build | DONE→m_city_check

  city 城池与街区:
    m_city_check    [🔀] → YES→m_city_plan | NO→结束
    m_city_plan     [🧠:location] → m_city_build
    m_city_build    [📝→location] → m_city_verify
    m_city_verify   [🧠] → CONTINUE→m_city_build | DONE→结束

👤 人设链

  c_n3_build_faction [📝→faction] → c_n2_check_faction  # 写完后回到判定

c_n4_number         [🧠] → c_n5a_gen_char           # 确定人数+名单

  # 角色循环
  c_n5a_gen_char    [🧠:character] → c_n5_build_char  # 生成属性
  c_n5_build_char   [📝→character] → c_n6_verify_template

  c_n6_verify_template [🧠] → PERFECT→c_n8 | NEEDS_FIX→c_n7_fix
  c_n7_fix          [📝→character:update] → c_n6_verify_template  # 补齐→重检

  c_n8_verify_quantity [🧠] → NEEDS_MORE→c_n5a | DONE→结束
